set(SPIRV)
if(PRECOMPILED_SHADER_DIR)
 if(NOT EXISTS "${PRECOMPILED_SHADER_DIR}/provenance.txt")
  message(FATAL_ERROR "Offline shader provenance.txt required")
 endif()
 file(READ "${PRECOMPILED_SHADER_DIR}/provenance.txt" PROVENANCE)
 string(FIND "${PROVENANCE}" "source_sha256=${SHADER_HASH}\n" SOURCE_HASH_POSITION)
 if(SOURCE_HASH_POSITION EQUAL -1)
  message(FATAL_ERROR "Offline shader source hash does not match resident.slang + exact_divide.slang")
 endif()
 file(GLOB PROVIDED_SPIRV "${PRECOMPILED_SHADER_DIR}/*.spv")
 list(LENGTH PROVIDED_SPIRV SPIRV_COUNT)
 if(NOT SPIRV_COUNT EQUAL 8)
  message(FATAL_ERROR "Exactly eight offline SPIR-V stage binaries required")
 endif()
 foreach(ENTRY IN LISTS ENTRIES)
  set(INPUT "${PRECOMPILED_SHADER_DIR}/${ENTRY}.spv")
  if(NOT EXISTS "${INPUT}" OR NOT EXISTS "${INPUT}.sha256")
   message(FATAL_ERROR "Missing offline stage or digest: ${ENTRY}")
  endif()
  file(SHA256 "${INPUT}" ACTUAL_HASH)
  file(READ "${INPUT}.sha256" EXPECTED_HASH)
  if(NOT EXPECTED_HASH STREQUAL "${ACTUAL_HASH}  ${ENTRY}.spv\n")
   message(FATAL_ERROR "Offline SPIR-V digest mismatch: ${ENTRY}")
  endif()
  configure_file("${INPUT}" "${SHADER_DIR}/${ENTRY}.spv" COPYONLY)
  configure_file("${INPUT}.sha256" "${SHADER_DIR}/${ENTRY}.spv.sha256" COPYONLY)
  list(APPEND SPIRV "${SHADER_DIR}/${ENTRY}.spv")
 endforeach()
 configure_file("${PRECOMPILED_SHADER_DIR}/provenance.txt" "${SHADER_DIR}/provenance.txt" COPYONLY)
else()
 if(NOT EXISTS "${SLANGC_EXECUTABLE}")
  message(FATAL_ERROR "Set SLANGC_EXECUTABLE or validated PRECOMPILED_SHADER_DIR; runtime compilation forbidden")
 endif()
 execute_process(COMMAND "${SLANGC_EXECUTABLE}" -version OUTPUT_VARIABLE SLANG_VERSION ERROR_VARIABLE SLANG_VERSION_ERROR OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE SLANG_RESULT)
 if(NOT SLANG_RESULT EQUAL 0)
  message(FATAL_ERROR "Slang compiler version probe failed")
 endif()
 file(WRITE "${SHADER_DIR}/provenance.txt" "compiler=${SLANGC_EXECUTABLE}\nversion=${SLANG_VERSION}${SLANG_VERSION_ERROR}\nsource_sha256=${SHADER_HASH}\nresident.slang=${RESIDENT_PARENT_HASH}\nexact_divide.slang=${EXACT_DIVIDE_HASH}\nflags=-target spirv -profile sm_6_0 -fp-mode precise -O2\n")
 foreach(ENTRY IN LISTS ENTRIES)
  set(OUTPUT "${SHADER_DIR}/${ENTRY}.spv")
  add_custom_command(OUTPUT "${OUTPUT}" "${OUTPUT}.sha256"
   COMMAND "${SLANGC_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/resident.slang" -entry "${ENTRY}" -stage compute -target spirv -profile sm_6_0 -fp-mode precise -O2 -o "${OUTPUT}"
   COMMAND "${CMAKE_COMMAND}" "-DSPIRV=${OUTPUT}" -P "${CMAKE_CURRENT_SOURCE_DIR}/shaders/hash_shader.cmake"
   DEPENDS shaders/resident.slang shaders/exact_divide.slang shaders/hash_shader.cmake VERBATIM)
  list(APPEND SPIRV "${OUTPUT}" "${OUTPUT}.sha256")
 endforeach()
endif()
