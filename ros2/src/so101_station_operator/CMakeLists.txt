cmake_minimum_required(VERSION 3.16)
project(so101_station_operator)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(so101_station_workflow REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/OperatorMaintenance.srv")
add_library(operator_panel SHARED src/operator_panel.cpp include/so101_station_operator/operator_panel.hpp src/ev3_panel.cpp include/so101_station_operator/ev3_panel.hpp)
target_include_directories(operator_panel PRIVATE include)
ament_target_dependencies(operator_panel rviz_common pluginlib rclcpp std_srvs std_msgs so101_station_workflow)
target_link_libraries(operator_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS operator_panel LIBRARY DESTINATION lib)
install(PROGRAMS scripts/operator_api DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  add_executable(ev3_panel_load_check test/ev3_panel_load.cpp)
  ament_target_dependencies(ev3_panel_load_check rviz_common pluginlib rclcpp std_msgs std_srvs so101_station_workflow)
  target_link_libraries(ev3_panel_load_check Qt5::Widgets)
  add_executable(operator_panel_load_check test/panel_load.cpp)
  ament_target_dependencies(operator_panel_load_check rviz_common pluginlib rclcpp)
  target_link_libraries(operator_panel_load_check Qt5::Widgets)
endif()

ament_package()
